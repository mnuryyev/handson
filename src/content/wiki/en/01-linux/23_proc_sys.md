/proc and /sys are virtual filesystems in the Linux kernel. There are no files on disk - everything is generated by the kernel at the moment of reading. Through them you can read kernel state, process information, hardware details, and change kernel parameters while the system is running.

    /proc  - procfs.  Information about processes and the kernel. Historically first.
             Mounted automatically at boot.
    /sys   - sysfs.   Structured information about devices and drivers.
             Introduced in kernel 2.6, more organized than /proc.

    Neither consumes disk space.
    Everything you read is generated by the kernel on the fly.
    Writing to certain files changes kernel behavior immediately.

## /proc - Structure Overview

    /proc/
    ├── 1/                  # directory for process PID 1 (init/systemd)
    ├── 2/                  # PID 2
    ├── .../                # one directory per process
    ├── cpuinfo             # CPU information
    ├── meminfo             # memory usage
    ├── net/                # network information
    ├── sys/                # kernel parameters (sysctl)
    ├── mounts              # mounted filesystems
    ├── filesystems         # supported filesystems
    ├── interrupts          # hardware interrupts
    ├── ioports             # I/O port map
    ├── iomem               # device memory map
    ├── modules             # loaded kernel modules
    ├── version             # kernel version
    ├── uptime              # system uptime
    ├── loadavg             # load average
    ├── stat                # CPU and interrupt statistics
    ├── diskstats           # disk statistics
    ├── swaps               # swap partitions
    ├── partitions          # disk partitions
    ├── devices             # registered devices
    ├── cmdline             # kernel boot parameters
    ├── kallsyms            # kernel symbols
    └── kcore               # kernel memory image (for gdb)

## /proc: Process Information

Every process has a directory /proc/<PID>/ with its own information.

    /proc/<PID>/
    ├── cmdline     # command line (arguments separated by \0)
    ├── status      # state, UID/GID, memory usage
    ├── stat        # statistics (used by ps, top)
    ├── statm       # memory usage in pages
    ├── maps        # virtual memory map
    ├── smaps       # detailed memory map (RSS, PSS, Swap)
    ├── fd/         # open file descriptors (symlinks)
    ├── fdinfo/     # file descriptor metadata
    ├── exe         # symlink to the executable
    ├── cwd         # symlink to the current working directory
    ├── root        # symlink to the process root (chroot)
    ├── environ     # environment variables (separated by \0)
    ├── io          # process I/O statistics
    ├── net/        # process network connections (in its namespace)
    ├── ns/         # process namespaces
    ├── limits      # resource limits (ulimit)
    ├── oom_score   # OOM killer score (0-1000)
    ├── oom_adj     # OOM killer adjustment (deprecated)
    ├── oom_score_adj  # OOM killer adjustment (-1000 to 1000)
    ├── wchan       # kernel function where process sleeps
    ├── stack       # kernel stack of the process
    └── task/       # threads (one directory per thread)

### Reading Process Information

    # Command line of a process
    cat /proc/1234/cmdline | tr '\0' ' '
    # \0 separates arguments → tr replaces with spaces

    # Environment variables of a process
    cat /proc/1234/environ | tr '\0' '\n'

    # Open files of a process
    ls -la /proc/1234/fd
    # lrwxrwxrwx 1 root root 0 ... 0 -> /dev/null
    # lrwxrwxrwx 1 root root 0 ... 1 -> /var/log/app.log
    # lrwxrwxrwx 1 root root 0 ... 7 -> socket:[12345]

    # Executable file of the process
    readlink /proc/1234/exe
    # /usr/bin/python3

    # Current working directory
    readlink /proc/1234/cwd

    # Process status
    cat /proc/1234/status
    # Name:   nginx
    # State:  S (sleeping)
    # Tgid:   1234       ← thread group ID = PID
    # Pid:    1234
    # PPid:   1          ← parent process
    # VmRSS:  4096 kB    ← actual RAM used
    # VmSize: 65536 kB   ← virtual memory size
    # Threads: 4          ← number of threads
    # voluntary_ctxt_switches:    100
    # nonvoluntary_ctxt_switches: 50

    # Process I/O statistics
    cat /proc/1234/io
    # rchar:  123456789   ← bytes read (including cache)
    # wchar:  987654321   ← bytes written
    # read_bytes:  65536  ← bytes actually read from disk
    # write_bytes: 32768  ← bytes actually written to disk

    # Virtual memory map
    cat /proc/1234/maps
    # 7f1234560000-7f1234570000 r-xp 00000000 08:01 123456 /usr/lib/libc.so.6
    # Fields: start-end perms offset device inode file

    # Detailed memory map (real RSS per region)
    cat /proc/1234/smaps | grep -A 10 "heap"

    # OOM killer: higher score = more likely to be killed
    cat /proc/1234/oom_score      # current score (0-1000)
    cat /proc/1234/oom_score_adj  # current adjustment

    # Protect process from OOM killer
    echo -1000 > /proc/1234/oom_score_adj
    # Make it candidate #1 to be killed
    echo 1000 > /proc/1234/oom_score_adj

    # Process resource limits (ulimit)
    cat /proc/1234/limits
    # Limit                     Soft Limit  Hard Limit  Units
    # Max open files            1024        4096        files
    # Max processes             31234       31234       processes

    # /proc/self - always refers to the current process
    cat /proc/self/status
    readlink /proc/self/exe

## /proc/sys - Kernel Parameters (sysctl)

/proc/sys contains kernel parameters changeable at runtime. Each file is one parameter.

    /proc/sys/
    ├── kernel/     # kernel settings (hostname, limits, panic)
    ├── net/        # network parameters
    ├── vm/         # memory management (Virtual Memory)
    ├── fs/         # filesystem settings
    └── dev/        # device parameters

    # Read a parameter directly
    cat /proc/sys/net/ipv4/ip_forward

    # Write a parameter directly
    echo 1 > /proc/sys/net/ipv4/ip_forward

    # Via sysctl (recommended)
    sysctl net.ipv4.ip_forward         # read
    sysctl -w net.ipv4.ip_forward=1    # write
    sysctl -a                          # all parameters
    sysctl -a | grep ipv4             # filter

    # Permanent changes - via /etc/sysctl.conf or /etc/sysctl.d/
    echo 'net.ipv4.ip_forward = 1' >> /etc/sysctl.conf
    sysctl -p   # apply from file

### Important kernel/ Parameters

    # Hostname
    cat /proc/sys/kernel/hostname
    echo "newhost" > /proc/sys/kernel/hostname

    # Kernel version
    cat /proc/sys/kernel/osrelease
    cat /proc/sys/kernel/version

    # Panic behavior
    cat /proc/sys/kernel/panic
    # 0 = hang on panic (default)
    # N = reboot after N seconds
    sysctl -w kernel.panic=10   # reboot after 10 seconds

    # Panic on oops (kernel error)
    sysctl -w kernel.panic_on_oops=1

    # Maximum open files system-wide
    cat /proc/sys/fs/file-max
    sysctl -w fs.file-max=2097152

    # Current file descriptor usage
    cat /proc/sys/fs/file-nr
    # open  reserved  maximum
    # 12345  0         2097152

    # Maximum PID
    cat /proc/sys/kernel/pid_max
    sysctl -w kernel.pid_max=4194304

    # dmesg message level
    cat /proc/sys/kernel/printk
    # current default minimum boot
    # 4       4       1       7

    # Magic SysRq (emergency kernel commands)
    cat /proc/sys/kernel/sysrq
    # 0 = disabled, 1 = all commands, N = bitmask
    sysctl -w kernel.sysrq=1
    # Usage: echo b > /proc/sysrq-trigger (immediate reboot)
    # echo s > /proc/sysrq-trigger (sync disks)
    # echo u > /proc/sysrq-trigger (remount read-only)

### Important net/ Parameters

    # IPv4 forwarding (routing between interfaces)
    cat /proc/sys/net/ipv4/ip_forward
    sysctl -w net.ipv4.ip_forward=1

    # IP spoofing protection (rp_filter)
    cat /proc/sys/net/ipv4/conf/all/rp_filter
    # 0 = disabled, 1 = strict, 2 = loose
    sysctl -w net.ipv4.conf.all.rp_filter=1

    # SYN cookies (SYN flood protection)
    sysctl -w net.ipv4.tcp_syncookies=1

    # Connection queues
    sysctl -w net.ipv4.tcp_max_syn_backlog=4096   # SYN backlog
    sysctl -w net.core.somaxconn=65535            # accept backlog

    # TIME_WAIT acceleration
    sysctl -w net.ipv4.tcp_tw_reuse=1    # reuse TIME_WAIT sockets

    # Keepalive
    sysctl -w net.ipv4.tcp_keepalive_time=600      # seconds before first keepalive
    sysctl -w net.ipv4.tcp_keepalive_intvl=60      # interval between keepalives
    sysctl -w net.ipv4.tcp_keepalive_probes=5      # probes before disconnect

    # TCP buffer sizes
    sysctl -w net.core.rmem_max=16777216           # max receive buffer
    sysctl -w net.core.wmem_max=16777216           # max send buffer
    sysctl -w net.ipv4.tcp_rmem="4096 87380 16777216"
    sysctl -w net.ipv4.tcp_wmem="4096 65536 16777216"

    # Local port range
    cat /proc/sys/net/ipv4/ip_local_port_range
    # 32768 60999 (default)
    sysctl -w net.ipv4.ip_local_port_range="1024 65535"

    # ARP table size
    sysctl -w net.ipv4.neigh.default.gc_thresh3=8192

    # IPv6 forwarding
    sysctl -w net.ipv6.conf.all.forwarding=1

    # Disable IPv6 (if not needed)
    sysctl -w net.ipv6.conf.all.disable_ipv6=1
    sysctl -w net.ipv6.conf.default.disable_ipv6=1

### Important vm/ Parameters

    # Swappiness: aggressiveness of swap usage (0-100)
    cat /proc/sys/vm/swappiness
    # 60 = default
    sysctl -w vm.swappiness=10   # for servers (less swap)

    # Dirty pages: when to flush cache to disk
    sysctl -w vm.dirty_ratio=10           # % RAM when process starts writing itself
    sysctl -w vm.dirty_background_ratio=5 # % RAM when flusher starts background write

    # OOM killer
    sysctl -w vm.panic_on_oom=0      # 0 = kill process, 1 = panic
    sysctl -w vm.overcommit_memory=1 # 0=heuristic, 1=always allow, 2=strict

    # Transparent Huge Pages
    cat /sys/kernel/mm/transparent_hugepage/enabled
    # [always] madvise never
    echo madvise > /sys/kernel/mm/transparent_hugepage/enabled

    # Minimum free memory
    sysctl -w vm.min_free_kbytes=65536

## /sys - sysfs Structure

/sys is organized by kernel subsystems. More structured than /proc.

    /sys/
    ├── block/          # block devices (disks)
    ├── bus/            # buses (PCI, USB, I2C, platform)
    ├── class/          # device classes (net, block, input)
    ├── dev/            # devices by major:minor numbers
    ├── devices/        # tree of all system devices
    ├── firmware/       # firmware interfaces (ACPI, EFI)
    ├── fs/             # filesystems (cgroup, bpf, fuse)
    ├── kernel/         # kernel parameters (debug, mm, power)
    ├── module/         # loaded modules and their parameters
    └── power/          # power management

### /sys/block - Block Devices

    ls /sys/block/
    # sda  sdb  nvme0n1  vda ...

    # Disk information
    cat /sys/block/sda/size              # size in 512-byte sectors
    cat /sys/block/sda/queue/rotational  # 1=HDD, 0=SSD/NVMe
    cat /sys/block/sda/queue/scheduler   # I/O scheduler
    cat /sys/block/sda/device/model      # disk model
    cat /sys/block/sda/device/vendor     # manufacturer

    # Disk statistics
    cat /sys/block/sda/stat
    # reads_completed reads_merged sectors_read ms_read
    # writes_completed writes_merged sectors_written ms_write
    # io_in_progress ms_io ms_weighted_io

    # I/O scheduler
    cat /sys/block/sda/queue/scheduler
    # [mq-deadline] kyber bfq none
    echo mq-deadline > /sys/block/sda/queue/scheduler   # change

    # Read-ahead
    cat /sys/block/sda/queue/read_ahead_kb
    echo 256 > /sys/block/sda/queue/read_ahead_kb

    # Queue depth (nr_requests)
    cat /sys/block/sda/queue/nr_requests
    echo 128 > /sys/block/sda/queue/nr_requests

    # SSD: check TRIM support
    cat /sys/block/sda/queue/discard_granularity  # 0 = TRIM not supported

    # Block sizes
    cat /sys/block/sda/queue/physical_block_size   # physical (usually 4096)
    cat /sys/block/sda/queue/logical_block_size    # logical (usually 512)
    cat /sys/block/sda/queue/optimal_io_size       # optimal I/O size

### /sys/class/net - Network Interfaces

    ls /sys/class/net/
    # eth0  lo  wlan0  docker0  br0 ...

    # Interface information
    cat /sys/class/net/eth0/speed          # speed in Mbps
    cat /sys/class/net/eth0/duplex         # half/full
    cat /sys/class/net/eth0/operstate      # up/down/unknown
    cat /sys/class/net/eth0/carrier        # 1=cable connected, 0=no cable
    cat /sys/class/net/eth0/mtu            # MTU
    cat /sys/class/net/eth0/address        # MAC address
    cat /sys/class/net/eth0/tx_queue_len   # transmit queue length
    cat /sys/class/net/eth0/type           # interface type (1=Ethernet)

    # Interface statistics
    cat /sys/class/net/eth0/statistics/rx_bytes
    cat /sys/class/net/eth0/statistics/tx_bytes
    cat /sys/class/net/eth0/statistics/rx_errors
    cat /sys/class/net/eth0/statistics/tx_dropped

    # Queues and buffers
    ls /sys/class/net/eth0/queues/
    # rx-0  tx-0  tx-1  ...

    # Change MTU
    echo 9000 > /sys/class/net/eth0/mtu   # Jumbo frames

### /sys/class/power_supply - Battery/Power

    ls /sys/class/power_supply/
    # AC  BAT0  BAT1 ...

    cat /sys/class/power_supply/BAT0/status        # Charging/Discharging/Full
    cat /sys/class/power_supply/BAT0/capacity      # charge percentage
    cat /sys/class/power_supply/BAT0/energy_now    # current charge (uWh)
    cat /sys/class/power_supply/BAT0/energy_full   # full charge (uWh)
    cat /sys/class/power_supply/BAT0/voltage_now   # voltage (uV)
    cat /sys/class/power_supply/BAT0/current_now   # current (uA)

### /sys/devices - Device Tree

    # Full device tree of the system
    ls /sys/devices/

    # PCI devices
    ls /sys/bus/pci/devices/
    # 0000:00:00.0  0000:00:02.0  0000:00:1f.2 ...

    # PCI device information
    cat /sys/bus/pci/devices/0000:00:02.0/vendor    # 0x8086 = Intel
    cat /sys/bus/pci/devices/0000:00:02.0/device    # device ID
    cat /sys/bus/pci/devices/0000:00:02.0/class     # device class
    cat /sys/bus/pci/devices/0000:00:02.0/driver    # active driver (symlink)

    # USB devices
    ls /sys/bus/usb/devices/

    # Temperature (hwmon)
    ls /sys/class/hwmon/
    cat /sys/class/hwmon/hwmon0/name        # sensor name
    cat /sys/class/hwmon/hwmon0/temp1_input # temperature in millidegrees
    # temp1_input = 45000 → 45°C

### /sys/kernel - Kernel Parameters

    # cgroups v2
    cat /sys/fs/cgroup/cgroup.controllers   # available controllers
    cat /sys/fs/cgroup/memory.usage_in_bytes

    # BPF
    ls /sys/fs/bpf/

    # Huge Pages
    cat /sys/kernel/mm/hugepages/hugepages-2048kB/nr_hugepages
    cat /sys/kernel/mm/hugepages/hugepages-2048kB/free_hugepages
    # Allocate huge pages:
    echo 128 > /sys/kernel/mm/hugepages/hugepages-2048kB/nr_hugepages

    # NUMA topology
    ls /sys/devices/system/node/
    cat /sys/devices/system/node/node0/meminfo
    cat /sys/devices/system/node/node0/cpulist   # which CPUs are in this NUMA node

    # CPU online/offline (disable a core)
    cat /sys/devices/system/cpu/cpu1/online   # 1=enabled
    echo 0 > /sys/devices/system/cpu/cpu1/online  # disable CPU1
    echo 1 > /sys/devices/system/cpu/cpu1/online  # re-enable

    # CPU frequency scaling
    ls /sys/devices/system/cpu/cpu0/cpufreq/
    cat /sys/devices/system/cpu/cpu0/cpufreq/scaling_governor
    # performance, powersave, ondemand, schedutil ...
    echo performance > /sys/devices/system/cpu/cpu0/cpufreq/scaling_governor

    # CPU cache sizes
    cat /sys/devices/system/cpu/cpu0/cache/index0/size   # L1 cache
    cat /sys/devices/system/cpu/cpu0/cache/index2/size   # L2 cache
    cat /sys/devices/system/cpu/cpu0/cache/index3/size   # L3 cache

## Practical Examples

### System Monitoring

    # Memory summary
    awk '/MemTotal|MemAvailable|MemFree|Cached|SwapTotal|SwapFree/ \
      {printf "%-15s %d MB\n", $1, $2/1024}' /proc/meminfo

    # Top processes by RSS memory
    for pid in /proc/[0-9]*/status; do
        awk '/^(Name|VmRSS)/{printf "%s ", $2}' $pid
        echo
    done | sort -k2 -rn | head -10

    # All open sockets of a process
    ls -la /proc/1234/fd | grep socket
    # Find port by socket inode:
    grep $(ls -la /proc/1234/fd | awk '/socket/{print $NF}' | \
      grep -o '[0-9]*') /proc/net/tcp

    # Memory usage by all processes
    awk '{sum += $2} END {print sum/1024 " MB"}' \
      /proc/*/status 2>/dev/null

    # Processes in state D (waiting for I/O, uninterruptible)
    for pid in /proc/[0-9]*/stat; do
        state=$(awk '{print $3}' $pid 2>/dev/null)
        [ "$state" = "D" ] && cat ${pid%stat}cmdline | tr '\0' ' ' && echo
    done

### Performance Tuning

    # Quick tuning for a web server
    cat >> /etc/sysctl.d/99-webserver.conf << 'EOF'
    # Network
    net.ipv4.ip_forward = 0
    net.ipv4.tcp_syncookies = 1
    net.core.somaxconn = 65535
    net.ipv4.tcp_max_syn_backlog = 8192
    net.core.rmem_max = 16777216
    net.core.wmem_max = 16777216
    net.ipv4.tcp_rmem = 4096 87380 16777216
    net.ipv4.tcp_wmem = 4096 65536 16777216
    net.ipv4.tcp_tw_reuse = 1
    net.ipv4.ip_local_port_range = 1024 65535
    # Memory
    vm.swappiness = 10
    vm.dirty_ratio = 15
    vm.dirty_background_ratio = 5
    EOF
    sysctl -p /etc/sysctl.d/99-webserver.conf

    # SSD tuning
    echo mq-deadline > /sys/block/sda/queue/scheduler
    echo 0 > /sys/block/sda/queue/rotational    # tell kernel this is SSD

## Cheat Sheet

    /proc - processes:
      /proc/<PID>/cmdline     - command line (tr '\0' ' ')
      /proc/<PID>/status      - state, memory, UID
      /proc/<PID>/fd/         - open file descriptors
      /proc/<PID>/maps        - virtual memory map
      /proc/<PID>/io          - I/O statistics
      /proc/<PID>/environ     - environment variables
      /proc/<PID>/oom_score_adj - OOM priority (-1000 protected, 1000 first victim)
      /proc/self/             - current process

    /proc - system:
      /proc/cpuinfo           - CPU information
      /proc/meminfo           - memory (MemAvailable = actually available)
      /proc/loadavg           - load average
      /proc/uptime            - uptime in seconds
      /proc/stat              - CPU statistics (read by top/htop)
      /proc/diskstats         - disk statistics (read by iostat)
      /proc/net/dev           - interface statistics
      /proc/net/tcp           - TCP connections
      /proc/mounts            - mounted filesystems
      /proc/modules           - loaded modules (lsmod)
      /proc/interrupts        - hardware interrupts

    /proc/sys - kernel parameters:
      net.ipv4.ip_forward=1                    - enable routing
      net.ipv4.tcp_syncookies=1                - SYN flood protection
      net.core.somaxconn=65535                 - connection backlog
      vm.swappiness=10                         - reduce swap usage
      kernel.panic=10                          - reboot on panic
      fs.file-max=2097152                      - max file descriptors

    /sys - devices:
      /sys/block/sda/queue/scheduler           - I/O scheduler
      /sys/block/sda/queue/rotational          - 0=SSD, 1=HDD
      /sys/class/net/eth0/speed                - interface speed
      /sys/class/net/eth0/statistics/rx_bytes  - traffic counters
      /sys/class/hwmon/hwmon0/temp1_input      - temperature (millidegrees)
      /sys/devices/system/cpu/cpu0/cpufreq/scaling_governor  - CPU governor
      /sys/kernel/mm/hugepages/hugepages-2048kB/nr_hugepages  - huge pages

    Reading and writing:
      cat /proc/sys/net/ipv4/ip_forward        - read
      echo 1 > /proc/sys/net/ipv4/ip_forward   - write (temporary)
      sysctl -w net.ipv4.ip_forward=1          - write via sysctl
      sysctl -p /etc/sysctl.conf               - apply from file
